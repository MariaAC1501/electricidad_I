\begin{frame}{Circuito RC: respuesta a un escalón}
    \begin{columns}[onlytextwidth]
    \begin{column}{0.5\textwidth}
        \centering
        \only<1->{
            \begin{circuitikz}[scale=0.8]
            \draw
            (0,0)
                to [V, l=$Vu(t-t_0)$]
            (0,4) 
                to [R, l=$R$]
            (4,4)
                to[C,l=$C$, v=$v$,voltage shift=3]
            (4,0) 
                to[short]
            (0,0)
            ;
            \draw<2-4> 
                (4,4)
                    to[open,i=$i_R$]
                (3,4)
                (4,4)
                    to[open,i=$i_C$]
                (4,3)
            ;
            \draw [white] (-3,-2) rectangle (5,6.2);
        \end{circuitikz}
        }
    \end{column}
    \begin{column}{0.5\textwidth}
        \only<1>
        {Se asume que el capacitor tiene un voltaje inicial
        \begin{equation*}
            v(t_0) = V_0
        \end{equation*}
        }\only<2-4>
            {\onslide<2->
            {aplicando LCK tenemos que:
            \begin{equation*}
                i_C + i_R = 0
            \end{equation*}
            }\onslide<3->
            {y sabemos que para todo $t>t_0$: 
            \begin{equation*}
                i_C = C\,\dfrac{dv}{dt},\, i_R=\dfrac{v-V}{R}
            \end{equation*}
            }\onslide<4->
            {entonces:
            \begin{equation*}
                C\,\dfrac{dv}{dt} + \dfrac{v-V}{R} = 0
            \end{equation*}
            }
        }\only<5-7>
            {\onslide<5->
            {dividiendo entre $C\,$ y reordenando:
            \begin{equation*}
                \dfrac{dv}{dt} = -\dfrac{v-V}{RC}
            \end{equation*}
            }\onslide<6->{
            reordenando nuevamente:
            \begin{equation*}
                \dfrac{dv}{v-V} = -\dfrac{1}{RC}dt
            \end{equation*}
            }\onslide<7->{
            integrando a ambos lados:
            \begin{equation*}
                \int\limits_{v(t_0)}^{v(t)}\dfrac{1}{v-V}\,dv = \int\limits_{t_0}^{t}-\frac{1}{RC}\,dt
            \end{equation*}
            }
        }\only<8-10>
            {\onslide<8->
            {tenemos que:
            \begin{equation*}
                \ln(v-V)\,\bigg|_{v(t_0)}^{v(t)} = -\frac{1}{RC}t\,\bigg|_{t_0}^{t}
            \end{equation*}
            }\onslide<9->
            {evaluando:
            \begin{equation*}
                \ln(v(t)-V) - \ln (v(t_0)-V) = -\frac{1}{RC}(t - t_0)
            \end{equation*}
            }\onslide<10->
            {osea:
            \begin{equation*}
                \dfrac{\ln(v(t)-V)}{\ln (v(t_0)-V)} = -\frac{1}{RC}(t - t_0)
            \end{equation*}
            }
        }\only<11-12>
            {\onslide<11->
            {aplicamos la función exponencial a ambos lados
            \begin{equation*}
                \dfrac{v(t)-V}{v(t_0)-V} = e\,^{-(t - t_0)/RC}
            \end{equation*}
            }\onslide<12->
            {sea $\tau=RC\,$ y $\,v(t_0) = V_0$, obtenemos:
            \begin{equation*}
                v(t)=V + (V_0-V)e\,^{-(t - t_0)/ \tau}
            \end{equation*}
            }
        }\only<13-15>
            {\onslide<13->
            {\begin{equation*}
                v(t) =
                \begin{cases}
                    V_0, & t < t_0\\
                    V + (V_0-V)e\,^{-(t - t_0)/ \tau}, & t > t_0
                \end{cases}
            \end{equation*}
            }\onslide<14->
            {\begin{equation*}
                i(t) = C\dfrac{dv}{dt} 
            \end{equation*}
            }\onslide<15->
            {\begin{equation*}
                i(t) = 
                \begin{cases}
                    0, & t < t_0\\
                    \dfrac{(V-V_0)}{R}e\,^{-(t - t_0)/ \tau}, & t > t_0
                \end{cases}
            \end{equation*}
            }
        }\only<16-18>
            {\onslide<16->
            {Para $t>t_0$ y reordenando tenemos:\\[2pt]
            \begin{tikzpicture}[scale=1]
                \draw
                (0,0) node[right]{$v(t) = V_0e\,^{-(t - t_0)/ \tau} + V(1-e\,^{-(t - t_0)/ \tau})$}
                ;
                \draw<17-> [decorate,decoration={brace,amplitude=5pt},xshift=0pt,yshift=-7pt]
                (3.3,0) -- (1.3,0) node [black,midway,below,align=center,yshift=-0.1cm] {\footnotesize Respuesta \\ \footnotesize natural}
                ;
                \draw<18-> [decorate,decoration={brace,amplitude=5pt},xshift=0pt,yshift=-7pt]
                (6.7,0) -- (3.8,0) node [black,midway,below,align=center,yshift=-0.1cm] {\footnotesize Respuesta \\ \footnotesize forzada}
                ;
            \end{tikzpicture} 
            }
        }\only<19->
            {\onslide<19->
            {Esta respuesta completa puede expresarse como:\\[2pt]
            \begin{equation*}
                v(t) = v(\infty) + [v(t_0)-v(\infty)]e\,^{-(t - t_0)/ \tau}
            \end{equation*}
            }
        }
    \end{column}
\end{columns}
\end{frame}

\begin{frame}{Ejemplo}
    \begin{columns}[onlytextwidth]
    \begin{column}{0.5\textwidth}
        \centering
        \only<1->{
            \begin{circuitikz}[scale=0.8]
            \draw
            (0,0)
                to [V, l=$v_f$]
            (0,4) 
                to [R, l=\SI{1}{\kilo\ohm}]
            (4,4)
                to[C,l=\SI{1}{\milli\farad}, v=$v$,voltage shift=3,i>^=$i$]
            (4,0) 
                to[short]
            (0,0)
            ;
            \node [right] at (-1.1,-1){$v_f(t) = 10\cdot [u(t-1) - u(t-3)]$};
            \draw [white] (-1.2,-2) rectangle (5.7,6.2);
        \end{circuitikz}
        }
    \end{column}
    \begin{column}{0.5\textwidth}
        \only<1>
        {Objetivo:\\[4pt]
        Averiguar $v(t)\,$ e $i(t)$  para todo $t$
        }\only<2,5,9>
        {$v_f(t)=$\\
        \begin{tikzpicture}[scale=1]
            \begin{axis}[%
                width= 0.9\linewidth,
                height= 0.5\linewidth,
                axis lines=center,
                xlabel=$t$,
                ylabel=\empty,
                x label style={anchor=west},
                y label style={anchor=south},
                ytick={1},
                yticklabels={$10$},
                ymax=1.5, % or enlarge y limits=upper
                xticklabels={$1$,$3$},
                xtick={1,3},
                xmin=-1,
                xmax=4,
                ]
            \addplot+[no marks, thick] coordinates {(-1,0) (1,0) (1,1) (3,1) (3,0) (4,0)};
            \end{axis}
        \end{tikzpicture}
        }\only<3-4>
            {\onslide<3->
            {para $t<1$:
            \begin{equation*}
                v(t)=0           
            \end{equation*}
            }\onslide<4->
            {por lo tanto
            \begin{equation*}
                v(1)=0           
            \end{equation*}
            }
        }\only<6-8>
            {\onslide<6->
            {para $1\leq t < 3$:
            \begin{equation*}
                v(t)= v(\infty) + [v(t_0)-v(\infty)]e\,^{-(t - t_0)/ \tau}          
            \end{equation*}
            }\onslide<7->
            {pero $v(\infty) = 10\,$, $\tau=1\,$, $t_0 = 1\,$ y $v(1)=0$, entonces: 
            \begin{equation*}
                v(t)=10\left(1-e\,^{-(t - 1)}\right)         
            \end{equation*}
            }\onslide<8->
            {entonces: 
            \begin{equation*}
                v(3)=10\left(1-e\,^{-2}\right)=\SI{8.647}{\volt}      
            \end{equation*}
            }
        }\only<10-11>
            {\onslide<10->
            {para $t \geq 3$:
            \begin{equation*}
                v(t)= v(\infty) + [v(t_0)-v(\infty)]e\,^{-(t - t_0)/ \tau}          
            \end{equation*}
            }\onslide<11->
            {pero $v(\infty) = 0\,$, $\tau=1\,$, $t_0 = 3\,$ y $v(3)=\SI{8.647}{\volt}$, entonces: 
            \begin{equation*}
                v(t)=8.647e\,^{-(t - 3)}        
            \end{equation*}
            }
        }\only<12->
            {\onslide<12->
            {Finalmente para todo $t\,$ tenemos que:
            \begin{equation*}
                v(t) =
                \begin{cases}
                    0 &\si{\volt},\quad t < 1\\
                    10\left(1-e\,^{-(t-1)}\right) &\si{\volt},\quad 1 \leq t < 3\\
                    8.647e^{-(t-3)} &\si{\volt},\quad t \geq 3
                \end{cases}
            \end{equation*}
            \begin{equation*}
                i(t) =
                \begin{cases}
                    0 &\si{\milli\ampere},\quad t < 1\\
                    10e\,^{-(t-1)} &\si{\milli\ampere},\quad 1 < t < 3\\
                    -8.647e^{-(t-3)} &\si{\milli\ampere},\quad t > 3
                \end{cases}
            \end{equation*}
            }
        }
    \end{column}
\end{columns}
\end{frame}
