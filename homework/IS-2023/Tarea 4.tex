\textbf{Superposición}

\begin{enumerate}
    \item Aplique la técnica de superposición y determine el valor de la corriente $i_0$.
    \begin{center}
        \begin{circuitikz}[scale = 1.5]
            \draw
             (2,3)

               to[V,l=\SI{3}{\volt}, invert]

                to[short]

            ;
        \end{circuitikz}
    \end{center}
    
  

    \item Aplique la técnica de superposición y determine el aporte de voltaje por cada fuente sobre la resistencia de 2 \si{\Omega}.

        \begin{circuitikz}[scale = 1.2]

        \end{circuitikz}
    \end{center}


    \item Utilice superposición para obtener $v_1$  y $v_2$

    \end{enumerate}
